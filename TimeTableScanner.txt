  <script>
    pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

    let parsedResults = [];

    // Bảng trọng số để Sort theo thứ trong tuần
    const dayOrder = {
      'Thứ 2': 1,
      'Thứ 3': 2,
      'Thứ 4': 3,
      'Thứ 5': 4,
      'Thứ 6': 5,
      'Thứ 7': 6,
      'Chủ Nhật': 7
    };

    function cleanStr(str) {
      if (!str) return '';
      return str.replace(/[\uE000-\uF8FF]/g, ' ').replace(/\s+/g, ' ').trim();
    }

    async function processPdf(event) {
      const file = event.target.files[0];
      if (!file) return;

      const statusEl = document.getElementById('status');
      statusEl.textContent = "⏳ Đang phân tích tọa độ và bóc tách...";

      try {
        const arrayBuffer = await file.arrayBuffer();
        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;

        let allTextItems = [];

        for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
          const page = await pdf.getPage(pageNum);
          const textContent = await page.getTextContent();

          const items = textContent.items.map(item => ({
            text: cleanStr(item.str),
            x: Math.round(item.transform[4]),
            y: Math.round(item.transform[5])
          })).filter(i => i.text.length > 0);

          allTextItems.push(...items);
        }

        // 1. Trích xuất Metadata
        const fullString = allTextItems.map(i => i.text).join(' ');
        const nameMatch = fullString.match(/Sinh\s*viên:\s*([^|\n\r]+)/i);
        const semesterMatch = fullString.match(/(Học\s*kỳ\s*\d+\s*-\s*Năm\s*học\s*[\d\s-]+)/i);

        document.getElementById('studentName').textContent = nameMatch ? nameMatch[1].trim() : 'Nguyễn Đình Thiện';
        document.getElementById('semester').textContent = semesterMatch ? semesterMatch[1].trim() : 'Học kỳ 4 - Năm học 2025 - 2026';
        document.getElementById('metadataCard').classList.remove('hidden');

        // 2. Định vị cột (Thứ 2 -> Chủ Nhật)
        const dayNames = ['Thứ 2', 'Thứ 3', 'Thứ 4', 'Thứ 5', 'Thứ 6', 'Thứ 7', 'Chủ Nhật'];
        const dayCols = [];

        dayNames.forEach(day => {
          const found = allTextItems.filter(item => item.text.startsWith(day));
          if (found.length > 0) {
            const avgX = found.reduce((acc, cur) => acc + cur.x, 0) / found.length;
            dayCols.push({ name: day, x: avgX });
          }
        });

        dayCols.sort((a, b) => a.x - b.x);

        function getDayByX(x) {
          if (dayCols.length === 0) return 'Chưa rõ';
          for (let i = 0; i < dayCols.length; i++) {
            let leftBound = (i === 0) ? 0 : (dayCols[i-1].x + dayCols[i].x) / 2;
            let rightBound = (i === dayCols.length - 1) ? 99999 : (dayCols[i].x + dayCols[i+1].x) / 2;
            if (x >= leftBound && x < rightBound) {
              return dayCols[i].name;
            }
          }
          return dayCols[0].name;
        }

        // 3. Dùng "Phòng:" làm điểm neo bóc tách
        const roomAnchors = allTextItems.filter(item => item.text.includes('Phòng:'));
        parsedResults = [];

        roomAnchors.forEach(anchor => {
          const day = getDayByX(anchor.x);

          const cluster = allTextItems.filter(item => 
            getDayByX(item.x) === day && 
            Math.abs(item.y - anchor.y) <= 130
          );

          cluster.sort((a, b) => b.y - a.y);
          const clusterText = cluster.map(i => i.text).join(' ');

          // A. Thời gian
          let time = 'Chưa rõ';
          const timeMatch = clusterText.match(/([0-2]?\d:[0-5]\d)\s*[^0-9a-zA-Z]{1,8}\s*([0-2]?\d:[0-5]\d)/);
          if (timeMatch) {
            time = `${timeMatch[1]} - ${timeMatch[2]}`;
          }

          // B. Mã môn & Tên môn
          let subjectCode = 'N/A';
          let subjectName = 'Chưa nhận diện';
          const codeMatch = clusterText.match(/([A-Z]{3,4}\s*\d{3})/i);
          if (codeMatch) {
            subjectCode = codeMatch[1].toUpperCase();
          }

          const subjectNameMatch = clusterText.match(/([A-Za-zÀ-ỹ\s]{3,60})\s*\(\s*[A-Z]{3,4}\s*\d{3}\s*\)/i);
          if (subjectNameMatch) {
            subjectName = subjectNameMatch[1].trim();
          } else if (codeMatch) {
            const idx = clusterText.indexOf(codeMatch[0]);
            if (idx > 0) {
              let before = clusterText.substring(0, idx).replace(/[()]/g, '').trim();
              if (before.length > 2) subjectName = before;
            }
          }

          // C. Nhóm
          let group = 'N/A';
          const groupMatch = clusterText.match(/Nhóm:\s*([A-Za-z0-9]+)/i);
          if (groupMatch) {
            group = groupMatch[1];
          }

          // D. Phòng
          let room = 'ONLINE / Chưa rõ';
          const roomMatch = clusterText.match(/Phòng:\s*([A-Za-z0-9\.\-\s]+?)(?=\s*GV:|\s*Lab|\s*Phòng|\s*ONLINE|\s*\d{1,2}:|$)/i);
          if (roomMatch) {
            room = roomMatch[1].trim();
          }

          // E. Giảng viên
          let lecturer = 'Chưa rõ';
          const lecturerMatch = clusterText.match(/GV:\s*([A-Za-zÀ-ỹ\s]+?)(?=\s*\d{1,2}:|\s*Lab|\s*Phòng|\s*ONLINE|\s*$)/i);
          if (lecturerMatch) {
            lecturer = lecturerMatch[1].trim();
          }

          parsedResults.push({
            day,
            time,
            subjectCode,
            subjectName,
            group,
            room,
            lecturer
          });
        });

        // 4. SORT THEO THỨ TRONG TUẦN (Thứ 2 ➔ Thứ 3 ➔ ... ➔ Chủ Nhật)
        parsedResults.sort((a, b) => {
          const valA = dayOrder[a.day] || 99;
          const valB = dayOrder[b.day] || 99;
          
          if (valA !== valB) {
            return valA - valB;
          }
          // Nếu trùng Thứ, tiếp tục xếp theo giờ bắt đầu
          return a.time.localeCompare(b.time);
        });

        renderTable(parsedResults);
        statusEl.textContent = `✅ Bóc tách & Sort thành công ${parsedResults.length} buổi học!`;

      } catch (err) {
        console.error(err);
        statusEl.textContent = "❌ Lỗi khi phân tích file PDF!";
      }
    }

    function renderTable(data) {
      const tbody = document.getElementById('resultBody');
      tbody.innerHTML = '';

      data.forEach(row => {
        const tr = document.createElement('tr');
        tr.className = 'border-b hover:bg-slate-50 transition';
        tr.innerHTML = `
          <td class="p-3 font-semibold text-blue-600">${row.day}</td>
          <td class="p-3">
            <span class="bg-blue-100 text-blue-800 text-xs font-bold px-2.5 py-1 rounded border border-blue-300 font-mono">
              ${row.time}
            </span>
          </td>
          <td class="p-3 font-mono font-bold text-slate-800">${row.subjectCode}</td>
          <td class="p-3 font-medium text-slate-800">${row.subjectName}</td>
          <td class="p-3">${row.group}</td>
          <td class="p-3">${row.room}</td>
          <td class="p-3">${row.lecturer}</td>
        `;
        tbody.appendChild(tr);
      });

      document.getElementById('totalItems').textContent = data.length;
      document.getElementById('resultCard').classList.remove('hidden');
    }

    function copyJson() {
      if (parsedResults.length === 0) return;
      navigator.clipboard.writeText(JSON.stringify(parsedResults, null, 2));
      alert('Đã copy dữ liệu JSON chuẩn bị sẵn cho Spring Boot!');
    }
  </script>
